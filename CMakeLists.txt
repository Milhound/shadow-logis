cmake_minimum_required(VERSION 3.21.1)
project(ShadowLogis VERSION 0.12.0 LANGUAGES CXX)
add_compile_definitions(SHADOW_LOGIS_VERSION="${PROJECT_VERSION}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>"
                        "$<$<COMPILE_LANGUAGE:CXX>:/permissive->" "$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>")
endif()
function(shadowlogis_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4>")
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Sql Pdf PdfWidgets PrintSupport)

target_include_directories(shadowlogis_data PUBLIC native)
target_link_libraries(shadowlogis_data PUBLIC Qt6::Core Qt6::Sql)
shadowlogis_warnings(shadowlogis_data)
if(WIN32)
    target_link_libraries(shadowlogis_data PRIVATE advapi32)
endif()
add_library(shadowlogis_documents native/documents.cpp native/companyprofile.cpp native/ratingdialog.cpp native/dimensionsdialog.cpp native/sightdraft.cpp native/airwaybill.cpp native/billoflading.cpp native/marine.cpp native/dangerousgoods.cpp native/preview.cpp native/airlinelookup.cpp native/reference.cpp native/appearance.cpp)

target_link_libraries(shadow-logis PRIVATE shadowlogis_documents)
shadowlogis_warnings(shadow-logis)
if(WIN32)
    # Desktop launches have no console; the companion preserves migration/CLI I/O.
    set_target_properties(shadow-logis PROPERTIES WIN32_EXECUTABLE TRUE)
    qt_add_executable(shadow-logis-cli native/main.cpp)
    set_target_properties(shadow-logis-cli PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(shadow-logis-cli PRIVATE shadowlogis_documents)
    shadowlogis_warnings(shadow-logis-cli)
    configure_file(native/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    target_sources(shadow-logis PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
    target_sources(shadow-logis-cli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()

    # End-user machines use Microsoft's official redistributable; do not copy
    # arbitrary MSVC runtime DLLs out of the developer's compiler installation.
    qt_generate_deploy_app_script(TARGET shadow-logis OUTPUT_SCRIPT deploy_script
        NO_COMPILER_RUNTIME NO_UNSUPPORTED_PLATFORM_ERROR)

    qt_generate_deploy_app_script(TARGET shadow-logis OUTPUT_SCRIPT deploy_script
        NO_UNSUPPORTED_PLATFORM_ERROR)
endif()
